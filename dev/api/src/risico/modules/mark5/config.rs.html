1use super::functions::kbdi_output;
2use crate::models::output::OutputElement;
3use crate::modules::kbdi::functions::kbdi_update_mm;

5/// configuration structure for model config
6/// can be used to store functions and constants
7#[derive(Debug)]
8pub struct Mark5ModelConfig {
9    pub model_version: String,
10    // soil moisture deficit function
11    smd_fn: fn(f32, f32, &[f32], f32) -> f32,
12    // return output element
13    get_output_fn: fn(f32, f32, f32, f32, f32, f32, f32) -> OutputElement,
14}
15
16impl Mark5ModelConfig {
17    pub fn new(model_version_str: &str) -> Self {
18        let smd_fn: fn(f32, f32, &[f32], f32) -> f32;
19        let get_output_fn: fn(f32, f32, f32, f32, f32, f32, f32) -> OutputElement;
20        match model_version_str {
21            "legacy" => {
22                smd_fn = kbdi_update_mm;
23                get_output_fn = kbdi_output;
24            }
25            _ => {
26                smd_fn = kbdi_update_mm;
27                get_output_fn = kbdi_output;
28            }
29        }
30
31        Mark5ModelConfig {
32            model_version: model_version_str.to_owned(),
33            smd_fn,
34            get_output_fn,
35        }
36    }
37
38    #[allow(non_snake_case, clippy::too_many_arguments)]
39    pub fn update_smd(
40        &self,
41        smd: f32,              // previous SMD value
42        temp: f32,             // temperature [°C]
43        history_rain: &[f32],  // daily rain of the last days [mm]
44        mean_annual_rain: f32, // mean annual rain [mm]
45    ) -> f32 {
46        (self.smd_fn)(smd, temp, history_rain, mean_annual_rain)
47    }
48
49    #[allow(non_snake_case, clippy::too_many_arguments)]
50    pub fn get_output(
51        &self,
52        smd: f32,
53        df: f32,
54        ffdi: f32,
55        temperature: f32,
56        rain: f32,
57        wind_speed: f32,
58        humidity: f32,
59    ) -> OutputElement {
60        (self.get_output_fn)(smd, df, ffdi, temperature, rain, wind_speed, humidity)
61    }
62}
