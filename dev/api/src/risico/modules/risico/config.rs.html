6type RosFnType = fn(f32, f32, f32, f32, f32, f32, f32, f32, f32, f32) -> (f32, f32);
7
8/// Default bounds, in hours, of the time step used to update the fine fuel
9/// moisture: the gridded runs are hourly or coarser.
10pub const DEFAULT_DT_BOUNDS_HOURS: (f32, f32) = (1.0, 72.0);
11
12/// configuration structure for model config
13/// can be used to store functions and constants
14#[derive(Debug)]
15pub struct RISICOModelConfig {
16    pub model_version: String,
17    pub use_t_effect: bool,
18    /// Bounds `(min, max)`, in hours, the time step between two inputs is
19    /// clamped to before updating the fine fuel moisture.
20    /// Defaults to [`DEFAULT_DT_BOUNDS_HOURS`]; point runs on sub-hourly
21    /// observations lower the minimum so a 10 minute step counts as 1/6 h.
22    pub dt_bounds_hours: (f32, f32),
23    ffmc_no_rain_fn: fn(f32, f32, f32, f32, f32, f32, f32) -> f32,
24    ffmc_rain_fn: fn(f32, f32, f32) -> f32,
25    ros_fn: RosFnType,
26    meteo_index_fn: fn(f32, f32) -> f32,
27}
28
29impl RISICOModelConfig {
30    /// The formulation for `model_version`: `legacy`, `v2023` or `v2025`.
31    /// Any other value falls back to `legacy`.

35        let ros_fn: RosFnType;
36        let meteo_index_fn: fn(f32, f32) -> f32;
37
38        match model_version_str {
39            "legacy" => {
40                ffmc_no_rain_fn = update_dffm_dry_legacy;

42                ros_fn = get_v_legacy;
43                meteo_index_fn = get_meteo_index_legacy;
44            }
45            // DEPRECATED
46            // "new_moisture" => {
47            //     ffmc_no_rain_fn = update_dffm_dry;
48            //     ffmc_rain_fn = update_dffm_rain;
49            //     ros_fn = get_v_legacy;
50            //     meteo_index_fn = get_meteo_index_legacy;
51            // }
52            // "new_ros" => {
53            //     ffmc_no_rain_fn = update_dffm_dry_legacy;
54            //     ffmc_rain_fn = update_dffm_rain_legacy;
55            //     ros_fn = get_v_v2023;
56            //     meteo_index_fn = get_meteo_index_legacy;
57            // }
58            "v2023" => {
59                ffmc_no_rain_fn = update_dffm_dry;
60                ffmc_rain_fn = update_dffm_rain;

62                meteo_index_fn = get_meteo_index_v2023;

68                meteo_index_fn = get_meteo_index_v2025;
69            }

71                ffmc_no_rain_fn = update_dffm_dry_legacy;

78        RISICOModelConfig {
79            model_version: model_version_str.to_owned(),

81            dt_bounds_hours: DEFAULT_DT_BOUNDS_HOURS,

83            ffmc_rain_fn,
84            ros_fn,
85            meteo_index_fn,
86        }
87    }

89    /// Set the bounds, in hours, of the moisture update time step.
90    pub fn with_dt_bounds_hours(mut self, min: f32, max: f32) -> Self {
91        assert!(
92            min >= 0.0 && min <= max,
93            "invalid dt bounds [{min}, {max}] hours"
94        );
95        self.dt_bounds_hours = (min, max);
96        self
97    }
98
99    /// Clamp a time step, in hours, to [`Self::dt_bounds_hours`].
100    pub fn clamp_dt_hours(&self, dt: f32) -> f32 {
101        dt.clamp(self.dt_bounds_hours.0, self.dt_bounds_hours.1)
102    }
103
104    #[allow(non_snake_case, clippy::too_many_arguments)]
105    pub fn ffmc_no_rain(

116    }
117
118    pub fn ffmc_rain(&self, r: f32, dffm: f32, sat: f32) -> f32 {

139    }
140
141    #[allow(non_snake_case, clippy::too_many_arguments)]
