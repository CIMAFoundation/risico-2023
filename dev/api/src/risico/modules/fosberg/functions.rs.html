4// Equilibrium Moisture Content [%] -> Simard formulation
5// From https://wikifire.wsl.ch/tiki-indexf2f0.html?page=Equilibrium+moisture+content
6pub fn emc(
7    temperature: f32, // temperature [°C]
8    humidity: f32,    // relative humidity [%]
9) -> f32 {
10    // conversion from °C to °F -> needed for the formula
11    let temp_f = temperature * 9.0 / 5.0 + 32.0;
12
13    if humidity < 10.0 {
14        0.03229 + 0.281073 * humidity - 0.000578 * temp_f * humidity
15    } else if humidity < 50.0 {
16        2.22749 + 0.160107 * humidity - 0.01478 * temp_f
17    } else {
18        21.0606 + 0.005565 * humidity.powi(2) - 0.00035 * temp_f * humidity - 0.483199 * humidity
19    }
20}

22// Fosberg Fire Weather Index
23// values range in [0 (no fire danger), 100 (high fire danger)]
24// some info: https://www.spc.noaa.gov/exper/firecomp/INFO/fosbinfo.html
25pub fn ffwi(
26    temperature: f32, // temperature [°C]
27    humidity: f32,    // relative humidity [%]
28    wind_speed: f32,  // wind speed [m/h]
29) -> f32 {
30    let emc = emc(temperature, humidity);
31    let ws_mph = wind_speed / 1609.344; // convert from m/h to mph
32    let emc_eff =
33        1.0 - 2.0 * (emc / 30.0) + 1.5 * (emc / 30.0).powi(2) - 0.5 * (emc / 30.0).powi(3);
34    let ffwi = emc_eff * f32::sqrt(1.0 + ws_mph.powi(2)) / 0.3002;
35    // clip in [0, 100]
36    ffwi.clamp(0.0, 100.0)
37}

39pub fn get_output_fn(state: &FosbergStateElement) -> OutputElement {
40    if (state.temp == NODATAVAL) || (state.humidity == NODATAVAL) || (state.wind_speed == NODATAVAL)
41    {
42        return OutputElement::default();
43    }
44    let ffwi = ffwi(state.temp, state.humidity, state.wind_speed);
45    let ws_out = state.wind_speed / 3600.0; // convert from m/h to m/s
46    OutputElement {
47        ffwi,                     // [-]
48        temperature: state.temp,  // [°C]
49        humidity: state.humidity, // [%]
50        wind_speed: ws_out,       // [m/s]
51        ..OutputElement::default()
52    }
53}
