21///Get the new value for the dfmm when is raining (p>p*)
22pub fn update_dffm_rain_legacy(r: f32, dffm: f32, sat: f32) -> f32 {
23    let delta_dffm = r
24        * R1_LEGACY
25        * f32::exp(-R2_LEGACY / ((sat + 1.0) - dffm))
26        * (1.0 - f32::exp(-R3_LEGACY / r));
27    let dffm = dffm + delta_dffm;
28
29    f32::min(dffm, sat)
30}

33///Get the new value for the dfmm when there is no rain (p<p*)
34pub fn update_dffm_dry_legacy(
35    dffm: f32,
36    _sat: f32,
37    T: f32,
38    W: f32,
39    H: f32,
40    T0: f32,
41    dT: f32,
42) -> f32 {
43    let EMC = A1_LEGACY * f32::powf(H, A2)
44        + A3 * f32::exp((H - 100.0) / 10.0)
45        + A4 * (30.0 - f32::min(T, 30.0)) * (1.0 - f32::exp(-A5 * H));
46    let K1 = T0 / (1.0 + A6 * f32::powf(T, B1) + A7 * f32::powf(W, B2));
47
48    // drying-wtting dynamic
49    let dffm = EMC + (dffm - EMC) * f32::exp(-dT / K1);
50
51    if dffm >= 0.0 {
52        dffm
53    } else {
54        0.0
55    }
56}

58///calculate PPF from the date and the two values
59pub fn get_ppf(time: &DateTime<Utc>, ppf_summer: f32, ppf_winter: f32) -> f32 {
60    const MARCH_31: u32 = 89;
61    const APRIL_1: u32 = 90;
62    const MAY_31: u32 = 150;
63    const JUNE_1: u32 = 151;
64    const SEPTEMBER_30: u32 = 272;
65    const OCTOBER_1: u32 = 273;
66    const NOVEMBER_30: u32 = 334;
67    const DECEMBER_1: u32 = 335;
68
69    if ppf_summer < 0.0 || ppf_winter < 0.0 {
70        return 0.0;
71    }
72    let day_number: u32 = time.date_naive().ordinal();
73
74    match day_number {
75        1..=MARCH_31 => ppf_winter,
76        APRIL_1..=MAY_31 => {
77            let val: f32 = (day_number - (MARCH_31 + 1)) as f32 / (MAY_31 - MARCH_31) as f32;
78            val * ppf_summer + (1.0 - val) * ppf_winter
79        }
80        JUNE_1..=SEPTEMBER_30 => ppf_summer,
81        OCTOBER_1..=NOVEMBER_30 => {
82            let val: f32 = 1.0
83                - ((day_number - (SEPTEMBER_30 + 1)) as f32 / (NOVEMBER_30 - SEPTEMBER_30) as f32);
84            val * ppf_summer + (1.0 - val) * ppf_winter
85        }
86        DECEMBER_1..=366 => ppf_winter,
87        _ => panic!("Invalid day number"),
88    }
89}

91///calculate the wind effect on fire propagation
92pub fn get_wind_effect_legacy(wind_speed: f32, wind_dir: f32, slope: f32, aspect: f32) -> f32 {
93    if wind_speed == NODATAVAL || wind_dir == NODATAVAL {
94        return 1.0;
95    }
96    //wind speed effect
97    let ws = (1.0 + DELTA1 * (DELTA2 + f32::tanh((wind_speed / DELTA3) - DELTA4)))
98        * (1.0 - (wind_speed / DELTA5));
99    let eta = wind_dir - aspect;
100    //aspect contribution
101    let mut n =
102        1.0 + (slope / (PI / 2.0)) * (ws - 1.0) * f32::exp(-f32::powf(eta - PI, 2.0) / QEPSIX2);
103    if n < 1.0 {
104        n = 1.0;
105    }
106    ws / n
107}

128///calculate the rate of spread
129#[allow(clippy::too_many_arguments)]
130pub fn get_v_legacy(
131    v0: f32,
132    d0: f32,
133    _d1: f32,
134    snow_cover: f32,
135    dffm: f32,
136    slope: f32,
137    aspect: f32,
138    wind_speed: f32,
139    wind_dir: f32,
140    t_effect: f32,
141) -> (f32, f32) {
142    let w_effect: f32 = get_wind_effect_legacy(wind_speed, wind_dir, slope, aspect);
143    if snow_cover > SNOW_COVER_THRESHOLD || d0 == NODATAVAL {
144        return (0.0, w_effect);
145    }
146
147    if dffm == NODATAVAL {
148        return (0.0, w_effect);
149    }
150
151    let moist_eff: f32 = get_moisture_effect_legacy(dffm);
152    let s_effect: f32 = get_slope_effect_legacy(slope);
153    let ros = v0 * moist_eff * w_effect * s_effect * t_effect;
154    (ros, w_effect)
155}

176///calculate the fire intensity
177pub fn get_intensity(
178    d0: f32,
179    d1: f32,
180    v: f32,
181    relative_greenness: f32,
182    lhv_dff: f32,
183    lhv_l1: f32,
184) -> f32 {
185    let mut d0 = d0;
186    let mut d1 = d1;
187
188    if d1 == NODATAVAL {
189        d1 = 0.0;
190    }
191    if d0 == NODATAVAL {
192        d0 = 0.0;
193    }
194
195    if relative_greenness >= 0.0 {
196        if d1 == 0.0 {
197            return v * (lhv_dff * d0 * (1.0 - relative_greenness)) / 3600.0;
198        }
199
200        return v * (lhv_dff * d0 + lhv_l1 * (d1 * (1.0 - relative_greenness))) / 3600.0;
201    }
202
203    v * (lhv_dff * d0 + lhv_l1 * d1) / 3600.0
204}

261///Get the new value for the dfmm when there is no rain (p<p*)
262#[allow(non_snake_case)]
263pub fn update_dffm_dry(dffm: f32, _sat: f32, T: f32, W: f32, H: f32, T0: f32, dT: f32) -> f32 {
264    let W = W / 3600.0; //wind is in m/h, should be in m/s
265
266    let EMC = A1 * f32::powf(H, A2)
267        + A3 * f32::exp((H - 100.0) / 10.0)
268        + A4 * (30.0 - f32::min(T, 30.0)) * (1.0 - f32::exp(-A5 * H));
269
270    let D_dry: f32 =
271        (1.0 + B1_D * f32::powf(T_STANDARD, C1_D) + B2_D * f32::powf(W_STANDARD, C2_D))
272            / (1.0 + B3_D * f32::powf(H_STANDARD, C3_D));
273    let K_dry: f32 = T0
274        * D_dry
275        * ((1.0 + B3_D * f32::powf(H, C3_D))
276            / (1.0 + B1_D * f32::powf(T, C1_D) + B2_D * f32::powf(W, C2_D)));
277
278    let D_wet: f32 = (1.0 + B3_W * f32::powf(H_STANDARD, C3_W))
279        / (1.0 + B1_W * f32::powf(T_STANDARD, C1_W) + B2_W * f32::powf(W_STANDARD, C2_W));
280    let K_wet: f32 = T0
281        * D_wet
282        * ((1.0 + B1_W * f32::powf(T, C1_W) + B2_W * f32::powf(W, C2_W))
283            / (1.0 + B3_W * f32::powf(H, C3_W)));
284
285    let K: f32 = if dffm >= EMC { K_dry } else { K_wet };
286
287    // drying-wtting dynamic
288    let const_G: f32 = (EMC - dffm) / (100.0 - dffm);
289    let dffm = (EMC - 100.0 * const_G * f32::exp(-dT / K)) / (1.0 - const_G * f32::exp(-dT / K));
290
291    if dffm >= 0.0 {
292        dffm
293    } else {
294        0.0
295    }
296}

301/// Get the wind effect on the fire propagation at the desired angle
302/// # Arguments
303/// * `wind_speed` - Wind speed \[m/h\]
304/// * `wind_dir` - Wind direction \[radians\]
305/// * `angle` - Angle \[radians\]
306/// # Returns
307/// * `w_eff_on_dir` - Wind effect in angle direction \[adim\]
308pub fn get_wind_effect_angle(wind_speed: f32, wind_dir: f32, angle: f32) -> f32 {
309    // convert from m/h to km/h
310    let mut ws_kph: f32 = wind_speed * 0.001;
311    // clip the value in [0, 60]
312    ws_kph = ws_kph.clamp(0.0, 60.0);
313    // constant for formula
314    let a_const: f32 = 1. - ((D1 * (D2 * f32::tanh((0. / D3) - D4))) + (0. / D5));
315    // contribution of wind - module
316    let w_eff_mod: f32 = a_const + (D1 * (D2 * f32::tanh((ws_kph / D3) - D4))) + (ws_kph / D5);
317    let a: f32 = (w_eff_mod - 1.) / 4.;
318    // normalize on direction
319    let theta: f32 = wind_dir - angle;
320    let theta_norm: f32 = (theta + PI) % (2. * PI) - PI;
321    let w_eff_on_dir: f32 = (a + 1.) * (1. - f32::powf(a, 2.)) / (1. - a * f32::cos(theta_norm));
322    w_eff_on_dir
323}

325/// Get the slope effect on the fire propagation at the desired angle
326/// # Arguments
327/// * `slope` - Slope \[radians\]
328/// * `aspect` - Aspect \[radians\]
329/// * `angle` - Angle \[radians\]
330/// # Returns
331/// * `h_eff_on_dir` - Slope effect in angle direction \[adim\]
332pub fn get_slope_effect_angle(slope: f32, aspect: f32, angle: f32) -> f32 {
333    // slope in angle direction
334    let s: f32 = f32::atan(f32::cos(aspect - angle) * f32::tan(slope));
335    // slope effect in angle direction
336    let h_eff_on_dir: f32 = f32::powf(2., f32::tanh(f32::powf(s * 3., 2.) * f32::signum(s)));
337    h_eff_on_dir
338}

340/// Get the combined effect of wind and slope on the fire propagation at the desired angle
341/// # Arguments
342/// * `slope` - Slope \[radians\]
343/// * `aspect` - Aspect \[radians\]
344/// * `wind_speed` - Wind speed \[m/h\]
345/// * `wind_dir` - Wind direction \[radians\]
346/// * `angle` - Angle \[radians\]
347/// # Returns
348/// * `wh` - Combined effect of wind and slope in angle direction \[adim\]
349pub fn get_wind_slope_effect_angle(
350    slope: f32,
351    aspect: f32,
352    wind_speed: f32,
353    wind_dir: f32,
354    angle: f32,
355) -> f32 {
356    let w_eff: f32 = get_wind_effect_angle(wind_speed, wind_dir, angle);
357    let s_eff: f32 = get_slope_effect_angle(slope, aspect, angle);
358    let wh: f32 = s_eff * w_eff;
359    wh
360}

362/// Get the wind and slope effect on the fire propagation considering all angles
363/// # Arguments
364/// * `slope` - Slope \[radians\]
365/// * `aspect` - Aspect \[radians\]
366/// * `wind_speed` - Wind speed \[m/h\]
367/// * `wind_dir` - Wind direction \[radians\]
368/// # Returns
369/// * `ws_effect` - Wind and slope effect \[adim\]
370pub fn get_wind_slope_effect(slope: f32, aspect: f32, wind_speed: f32, wind_dir: f32) -> f32 {
371    let angles: Array1<f32> = Array::linspace(0., 2. * PI, N_ANGLES_ROS);
372    let ws_effect: f32 = angles
373        .iter()
374        .map(|x| get_wind_slope_effect_angle(slope, aspect, wind_speed, wind_dir, *x))
375        .reduce(f32::max)
376        .unwrap_or(NODATAVAL);
377    ws_effect
378}

380pub fn get_moisture_effect_v2023(dffm: f32) -> f32 {
381    // normalize in [0, 1] and divide by moisture of extintion
382    let x: f32 = (dffm / 100.) / MX;
383    // moisture effect
384    let moist_eff: f32 = M5 * f32::powf(x, 5.)
385        + M4 * f32::powf(x, 4.)
386        + M3 * f32::powf(x, 3.)
387        + M2 * f32::powf(x, 2.)
388        + M1 * x
389        + M0;
390    // clip in [0, 1]
391    moist_eff.clamp(0.0, 1.)
392}

576        0.0
577    };
578    let r = if rain != NODATAVAL { rain } else { 0.0 };
579
580    if r > MAXRAIN {
581        state.dffm = config.ffmc_rain(r, state.dffm, sat);
582    } else {
583        state.dffm = config.ffmc_no_rain(state.dffm, sat, t, w, h, T0, dt);
584    }
585
586    // limit dffm to [0, sat]
587
588    state.dffm = f32::max(0.0, f32::min(sat, state.dffm));
589}

591
592#[allow(non_snake_case)]
593pub fn get_output_fn(
594    state: &RISICOStateElement,
595    props: &RISICOPropertiesElement,
596    input: &InputElement,
597    config: &RISICOModelConfig,
598    time: &DateTime<Utc>,
599) -> OutputElement {
600    let veg = &props.vegetation;
601
602    let wind_dir = input.wind_dir;
603    let wind_speed = input.wind_speed;
604    let humidity = input.humidity;
605    let rain = input.rain;
606
607    let slope = props.slope;
608    let aspect = props.aspect;
609
610    let temperature = input.temperature;
611    let snow_cover = state.snow_cover;
612    let NDVI = state.NDVI;
613    let NDWI = state.NDWI;
614
615    let dffm = state.dffm;
616
617    let ndvi = if veg.use_ndvi && NDVI != NODATAVAL {
618        (1.0 - NDVI).clamp(0.0, 1.0)
619    } else {
620        1.0
621    };
622
623    let ndwi = if NDWI != NODATAVAL {
624        (1.0 - NDWI).clamp(0.0, 1.0)
625    } else {
626        1.0
627    };
628
629    let t_effect = if config.use_t_effect {
630        get_t_effect(temperature)
631    } else {
632        1.0
633    };
634
635    let (ros, wind_effect) = config.ros(
636        veg.v0, veg.d0, veg.d1, dffm, snow_cover, slope, aspect, wind_speed, wind_dir, t_effect,
637    );
638
639    let meteo_index = config.meteo_index(dffm, wind_effect);
640
641    let ppf = get_ppf(time, props.ppf_summer, props.ppf_winter);
642
643    let intensity = if ros != NODATAVAL && veg.hhv != NODATAVAL {
644        let LHVdff = get_lhv_dff(veg.hhv, dffm);
645        // calcolo LHV per la vegetazione viva
646        let LHVl1 = get_lhv_l1(veg.umid, state.MSI, veg.hhv);
647        // Calcolo Intensità
648        get_intensity(veg.d0, veg.d1, ros, state.NDVI, LHVdff, LHVl1)
649    } else {
650        NODATAVAL
651    };
652
653    let wind_speed_out = wind_speed / 3600.0; // convert to m/s
654    let wind_dir_out = wind_dir.to_degrees();
655    OutputElement {
656        V: ros,
657        W: wind_effect,
658        PPF: ppf,
659        I: intensity,
660        temperature,
661        humidity,
662        wind_speed: wind_speed_out,
663        wind_dir: wind_dir_out,
664        rain,
665        snow_cover,
666        dffm,
667        t_effect,
668        NDWI: ndwi,
669        NDVI: ndvi,
670        meteo_index,
671        ..OutputElement::default()
672    }
673}
