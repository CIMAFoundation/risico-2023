4pub fn daylight_hours(
5    latitude: f32,  // latitude [°]
6    date: DateTime<Utc>, // datetime
7) -> f32 {
8    // Calculate the number of daylight hours for a given latitude and julian day
9    // Based on the FAO formula from https://wikifire.wsl.ch/tiki-index9a98.html?page=Daylight+hours&structure=Fire
10    let jday = date.ordinal() as f32;
11    let declination  = 0.409 * ((2.0 * PI / 365.0) * jday - 1.39).sin();
12    let latitude_rad = latitude * PI / 180.0;
13    let sunset_angle = (-(latitude_rad).tan() * (declination).tan()).acos();
14    
15    sunset_angle * 24.0 / PI
16}

19pub fn evapotranspiration_thornthwaite(
20    temp: f32,  // air temperature [°C]
21    latitude: f32,  // latitude [°]
22    date: DateTime<Utc>, // date
23    heat_index: f32, // heat index [°C]
24) -> f32 { // [mm]
25    // Thornthwaite method for calculating potential evapotranspiration
26    // Thornthwaite, C. W. (1948). An approach toward a rational classification of climate.
27    // Geographical Review, 38(1), 55-94.
28    // NOTE: the original formula uses mean daily ari temperature, but other modifications are possible
29    let a = 6.75e-7 * heat_index.powf(3.0)
30        - 7.71e-5 * heat_index.powf(2.0)
31        + 0.01792 * heat_index
32        + 0.49239;
33    let hsum: f32 = daylight_hours(latitude, date);
34    let pet: f32 = if temp < 0.0 {
35        0.0
36    } else if temp <= 26.0 {
37        16.0*(hsum/360.0)*(10.0*temp/heat_index).powf(a)
38    } else {
39        (hsum/360.0) * (-415.85 + 30.533224*temp - 0.43*temp.powf(2.0))
40    };
41    pet
42}

44pub fn heat_index(
45    monthly_mean_temp: Vec<f32>,  // monthly mean temperature [°C]
46) -> f32 {
47    // sum of the maximum between 0 and the monthlym mean temperature
48    let heat_index = monthly_mean_temp.iter()
49    .map(|&x| (x.max(0.0)/5.0).powf(1.514))
50    .sum();
51    heat_index
52}
