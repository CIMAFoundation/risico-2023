9pub fn store_day_fn(state: &mut KBDIStateElement, input: &InputElement) {
10    // cumulated rain per day
11    if input.rain > 0.0 {
12        state.cum_rain += input.rain;
13    }
14    // store the maximum temperature per day
15    if (state.max_temp == NODATAVAL) || (input.temperature > state.max_temp) {
16        state.max_temp = input.temperature;
17    }
18}

20// Keetch-Byram Drought Index
21// It is expressed as soil moisture deficit in mm, and ranges from 0 mm (wet) to 200 mm (dry)
22// Source: WikiFire, Finkele et al. 2006
23pub fn kbdi_update_mm(
24    kbdi: f32,             // previous KBDI value [mm]
25    max_temp: f32,         // maximum daily temperature [°C]
26    history_rain: &[f32], // daily rain of the last days + rain of today [mm] (ordered from oldest to newest)
27    mean_annual_rain: f32, // mean annual rain [mm]
28) -> f32 {
29    // rain of the day
30    let day_rain = history_rain[history_rain.len() - 1];
31    // calculate the rain of the last days (in case they are consecutive)
32    let mut last_rain = 0.0;
33    if history_rain.len() >= 2 {
34        let mut idx = history_rain.len() - 2; // starting from yesterday
35        while (idx > 0) && (history_rain[idx] > 0.0) {
36            last_rain += history_rain[idx];
37            idx -= 1;
38        }
39    }
40    // effective rain of the day
41    let effective_rain = f32::max(0.0, day_rain - f32::max(0.0, KBDI_RAIN_RUNOFF - last_rain));
42    let dt: f32 = 1.0; // DAILY COMPUTATION
43    let qsi = kbdi - effective_rain;
44    let evapo_transp: f32 =
45        (((203.2 - qsi) * (0.968 * f32::exp(0.0875 * max_temp + 1.5552) - 8.3) * dt)
46            / (1.0 + 10.88 * f32::exp(-0.001736 * mean_annual_rain)))
47            * 10e-3;
48    let kbdi_new = qsi + evapo_transp;
49    kbdi_new.clamp(0.0, 200.0)
50}

52pub fn update_fn(
53    state: &mut KBDIStateElement,
54    prop: &KBDIPropertiesElement,
55    config: &KBDIModelConfig,
56    time: &DateTime<Utc>,
57) {
58    // store the datetime and cumulated rain for the day of the run
59    state.update(time, state.cum_rain);
60    // get the last rains in the time windows -> they are already ordered from oldest to newest
61    let (_, daily_rains) = state.get_time_window(time);
62    if state.max_temp == NODATAVAL {
63        return; // no update
64    }
65    let new_kbdi = config.update_kbdi(state.kbdi, state.max_temp, &daily_rains, prop.mean_rain);
66    // store the new KBDI value
67    state.kbdi = new_kbdi;
68}
