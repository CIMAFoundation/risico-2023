17// Store the daily info at 15:00 local time
18pub fn store_day_fn(
19    state: &mut NesterovStateElement,
20    input: &InputElement,
21    prop: &NesterovPropertiesElement,
22    time: &DateTime<Utc>,
23) {
24    // cumulate rain
25    if input.rain > 0.0 {
26        state.cum_rain += input.rain;
27    }
28    // store the other daily info
29    let tz_name = TZ_FINDER.get_tz_name(prop.lon as f64, prop.lat as f64);
30    let tz : Tz = tz_name.parse().expect("Invalid timezone name");
31    let local_time = time.with_timezone(&tz);
32    if local_time.hour() <= TIME_WEATHER {  // save the last available data
33        state.temp_15 = input.temperature;
34        state.temp_dew_15 = input.temp_dew;
35    }
36}

39// Nesterov Ignition Index
40pub fn nesterov_update(
41    nesterov: f32,  // Nesterov Index [-]
42    temp_15: f32,  // temperature [°C] at 15:00
43    temp_dew_15: f32,  // dew point temperature [°C] at 15:00
44    daily_rain: f32,  // daily rain [mm]
45) -> f32 {
46    let new_nesterov: f32 = if daily_rain > RAIN_TH {
47        0.0
48    } else {
49        let nest = nesterov + temp_15*(temp_15 - temp_dew_15);
50        if nest < 0.0 {
51            0.0
52        } else {
53            nest
54        }
55    };
56    new_nesterov
57}

60// update the Nesterov index
61pub fn update_fn(
62    state: &mut NesterovStateElement,
63) {
64    if (state.temp_15 == NODATAVAL) || (state.temp_dew_15 == NODATAVAL) {
65        return  // no update
66    }
67    state.nesterov = nesterov_update(state.nesterov, state.temp_15, state.temp_dew_15, state.cum_rain);
68}
