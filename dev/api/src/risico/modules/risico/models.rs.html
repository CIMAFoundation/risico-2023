37/// The fuel parameters of a vegetation class: one row of the vegetation

40#[allow(non_snake_case)]
41#[derive(Debug)]
42pub struct RISICOVegetation {
43    /// Identifier, as in the vegetation-id layer
44    pub id: String,
45    /// Dead fine fuel load
46    pub d0: f32,
47    /// Live fuel load
48    pub d1: f32,
49    /// Higher heating value [kJ/kg]
50    pub hhv: f32,
51    /// Live fuel moisture
52    pub umid: f32,
53    /// Base rate of spread [m/h]
54    pub v0: f32,
55    /// Response time of the dead fine fuel moisture [h]
56    pub T0: f32,
57    /// Saturation moisture of the dead fine fuel [%]
58    pub sat: f32,
59    /// Description
60    pub name: String,
61    /// Whether satellite NDVI modulates the live fuel
62    pub use_ndvi: bool,
63}

131pub struct RISICOState {

133    pub time: DateTime<Utc>,
134    pub data: Array1<RISICOStateElement>,
135    len: usize,

139impl RISICOState {
140    /// Create the state of `warm_state.len()` cells at `time`.
141    #[allow(dead_code, non_snake_case)]
142    pub fn new(warm_state: &[RISICOWarmState], time: &DateTime<Utc>, config: RISICOModelConfig) -> RISICOState {
143        let data = Array1::from_vec(
144            warm_state
145                .iter()
146                .map(|w| RISICOStateElement {
147                    dffm: w.dffm,
148                    snow_cover: w.snow_cover,
149                    snow_cover_time: w.snow_cover_time,
150                    MSI: w.MSI,
151                    MSI_TTL: w.MSI_TTL,
152                    NDVI: w.NDVI,
153                    NDVI_TIME: w.NDVI_TIME,
154                    NDWI: w.NDWI,
155                    NDWI_TIME: w.NDWI_TIME,
156                })
157                .collect(),
158        );
159
160        RISICOState {
161            time: *time,
162            // props,
163            data,
164            len: warm_state.len(),
165            config,
166        }
167    }

177    fn update_snow_cover(&mut self, input: &Input) {
178        let time = input.time.timestamp() as f32;
179
180        Zip::from(&mut self.data)
181            .and(&input.data)
182            .for_each(|state, input| {
183                let i_snow_cover = input.snow_cover;
184
185                if i_snow_cover == NODATAVAL {
186                    if (time - state.snow_cover_time) as i64 > SNOW_SECONDS_VALIDITY {
187                        state.snow_cover = NODATAVAL;
188                    }
189                    return;
190                }
191
192                state.snow_cover = i_snow_cover;
193                state.snow_cover_time = time;
194            });
195    }
196
197    fn update_satellite(&mut self, input: &Input) {
198        let time = input.time.timestamp() as f32;
199        Zip::from(&mut self.data)
200            .and(&input.data)
201            .for_each(|state, input| {
202                let i_msi = input.msi;
203
204                if !(0.0..=1.0).contains(&i_msi) {
205                    if state.MSI > 0.0 {
206                        state.MSI_TTL -= 1.0;
207                    } else {
208                        state.MSI = NODATAVAL;
209                    }
210                } else {
211                    state.MSI = i_msi;
212                    state.MSI_TTL = 56.0;
213                }
214            });
215        Zip::from(&mut self.data)
216            .and(&input.data)
217            .for_each(|state, input| {
218                let i_ndvi = input.ndvi;
219
220                if self.time.timestamp() - state.NDVI_TIME as i64 > SATELLITE_DATA_SECONDS_VALIDITY
221                {
222                    state.NDVI = NODATAVAL;
223                }
224
225                if i_ndvi != NODATAVAL {
226                    if (0.0..=1.0).contains(&i_ndvi) {
227                        state.NDVI = i_ndvi;
228                    } else {
229                        state.NDVI = NODATAVAL;
230                    }
231
232                    state.NDVI_TIME = time;
233                }
234            });
235        Zip::from(&mut self.data)
236            .and(&input.data)
237            .for_each(|state, input| {
238                let i_ndwi = input.ndwi;
239
240                if self.time.timestamp() - state.NDWI_TIME as i64 > SATELLITE_DATA_SECONDS_VALIDITY
241                {
242                    state.NDWI = NODATAVAL;
243                }
244
245                if self.time.timestamp() - state.NDWI_TIME as i64 > 240 * 3600 {
246                    state.NDWI = NODATAVAL;
247                }
248
249                if i_ndwi != NODATAVAL {
250                    if (0.0..=1.0).contains(&i_ndwi) {
251                        state.NDWI = i_ndwi;
252                    } else {
253                        state.NDWI = NODATAVAL;
254                    }
255                    state.NDWI_TIME = time;
256                }
257            });
258    }

260    #[allow(non_snake_case)]
261    fn update_moisture(&mut self, props: &RISICOProperties, input: &Input, dt: f32) {
262        let dt = self.config.clamp_dt_hours(dt);
263
264        Zip::from(&mut self.data)
265            // .and(&self.snow_cover)
266            .and(&props.data)
267            .and(&input.data)
268            .for_each(|state, props, input_data| {
269                update_moisture_fn(state, props, input_data, &self.config, dt)
270            });
271    }
272
273    #[allow(non_snake_case)]
274    pub fn get_output(self: &RISICOState, props: &RISICOProperties, input: &Input) -> Output {
275        let time = &self.time;
276
277        let output_data = Zip::from(&self.data)
278            .and(&props.data)
279            .and(&input.data)
280            .map_collect(|state, props, input| {
281                get_output_fn(state, props, input, &self.config, time)
282            });
283
284        Output::new(*time, output_data)
285    }
286
287    /// Advance the state to `input.time`, applying the weather in `input`.
288    /// The time step is the difference from the previous update.
289    pub fn update(&mut self, props: &RISICOProperties, input: &Input) {
290        let new_time = &input.time;
291        let dt = new_time.signed_duration_since(self.time).num_seconds() as f32 / 3600.0;
292        self.time = *new_time;
293        self.update_satellite(input);
294        self.update_snow_cover(input);
295        self.update_moisture(props, input, dt);
296    }
297
298    /// Compute the outputs (`dffm`, `V`, `I`, `W`, `contrT`, ...) at the
299    /// current time. `input` must be the input of the last update.
300    pub fn output(&self, props: &RISICOProperties, input: &Input) -> Output {
301        self.get_output(props, input)
302    }

306#[derive(Debug)]
307pub struct RISICOProperties {
308    pub data: Array1<RISICOPropertiesElement>,
309    pub vegetations_dict: HashMap<String, Arc<RISICOVegetation>>,
310    pub len: usize,

313impl RISICOProperties {
314    /// Build the properties of `props.lons.len()` cells. The PPF vectors are
315    /// aligned with the cells; pass `1.0` where PPF is not used.
316    pub fn new(
317        props: RISICOCellPropertiesContainer,
318        vegetations_dict: HashMap<String, Arc<RISICOVegetation>>,
319        ppf_summer: Vec<f32>,
320        ppf_winter: Vec<f32>,
321    ) -> Self {
322        // check if all vectors have the same length
323        // let n_elements = lats.len();
324
325        // if ppf_summer.len() < n_elements {
326        //     println!("Warning: PPF is not consistent with cell file. Overriding missing elements.");
327        //     println!("PPF: {} elements ", ppf_summer.len());
328        //     println!("Cells: {} elements ", n_elements);
329
330        // }
331
332        // The vegetation of each cell is resolved while the static layers are
333        // read, so this holds one shared handle per cell instead of one owned
334        // identifier string.
335        let data: Array1<RISICOPropertiesElement> = props
336            .vegetations
337            .iter()
338            .enumerate()
339            .map(|(idx, vegetation)| RISICOPropertiesElement {
340                lon: props.lons[idx],
341                lat: props.lats[idx],
342                slope: props.slopes[idx],
343                aspect: props.aspects[idx],
344                ppf_summer: ppf_summer[idx],
345                ppf_winter: ppf_winter[idx],
346                vegetation: vegetation.clone(),
347            })
348            .collect();
349
350        let len = data.len();
351        Self {
352            data,
353            vegetations_dict,
354            len,
355        }
356    }

358    pub fn get_coords(&self) -> (Vec<f32>, Vec<f32>) {
359        let lats: Vec<f32> = self.data.iter().map(|p| p.lat).collect();
360        let lons: Vec<f32> = self.data.iter().map(|p| p.lon).collect();
361        (lats, lons)
362    }
